{The parent selects the nearest fixed state prototype using absolute distances, then adds a small persistence bonus. The main hypothesis is to replace those prototypes with relative dispersion scores: favor member 0 when unfinished-job work varies strongly early in construction, member 9 when machine end times vary strongly, and member 5 as the baseline, reflecting their respective work, waiting, and efficiency terms. The categorical persistence bonus remains; these state-to-member mappings are design hypotheses without evaluation evidence.}

def use_heuristics(progress, candidate_pair_ratio, remaining_work_mean, remaining_work_std, machine_ready_mean, machine_ready_std, previous_heuristic):
    work_dispersion = remaining_work_std / (
        remaining_work_mean + remaining_work_std + 0.000000000001
    )
    ready_dispersion = machine_ready_std / (
        machine_ready_mean + machine_ready_std + 0.000000000001
    )

    s0 = 0.15 + (1.0 - progress) * work_dispersion
    s5 = 0.45
    s9 = 0.15 + 0.70 * ready_dispersion

    if previous_heuristic == 0:
        s0 = s0 + 0.03
    if previous_heuristic == 5:
        s5 = s5 + 0.03
    if previous_heuristic == 9:
        s9 = s9 + 0.03

    return (0, (s0, -1.0, -1.0, -1.0, -1.0, s5, -1.0, -1.0, -1.0, s9))
